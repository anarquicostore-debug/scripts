    status.Text       = msg
    status.TextColor3 = Color3.fromRGB(r, g, b)

            if status.Text == msg then status.Text = "" end

-- ─── INVITE REMOTE ────────────────────────────────────────────────────────────
-- Origem: ReplicatedStorage.Controllers.TradeController
--   linha 43 : local v_u_40 = Net:RemoteFunction("TradeService/Invite")
--   linha 355: function TradeController.SendInvite(_, userId, onResult)
--              upvalues: (copy) v_u_40, (copy) v_u_138, (copy) v_u_139
--   linha 357: v_u_40:InvokeServer("8fbe1594-7cef-4c29-94d1-a0e93adfa5a4", userId)
-- A remote é lida dos upvalues do próprio módulo; não usamos o pacote Net.
local INVITE_GUID = "8fbe1594-7cef-4c29-94d1-a0e93adfa5a4"
local SEARCH_GUID = "5bbf0ed7-abce-44dc-aec6-d72992f7e476" -- TradeController:615
local SEARCH_COOLDOWN = 5 -- o jogo trava a busca por 5s (TradeController:629)

local getupvals = (debug and debug.getupvalues) or getupvalues

local TradeController
do
    local ok, mod = pcall(function()
        return require(ReplicatedStorage:WaitForChild("Controllers"):WaitForChild("TradeController"))
    end)
    if ok and type(mod) == "table" and type(mod.SendInvite) == "function" then
        TradeController = mod
    end
end

-- Retorna a única RemoteFunction entre os upvalues de fn (nil se houver 0 ou >1).
local function onlyRemoteFunction(fn)
    if not getupvals or type(fn) ~= "function" then return nil end
    local ok, ups = pcall(getupvals, fn)
    if not ok or type(ups) ~= "table" then return nil end
    local found
    for _, v in pairs(ups) do
        if typeof(v) == "Instance" and v:IsA("RemoteFunction") then
            if found and found ~= v then return nil end
            found = v
        end
    end
    return found
end

local inviteRemote = TradeController and onlyRemoteFunction(TradeController.SendInvite)

-- SearchUser: upvalue nº 12 de _createPlayerList (v_u_39). Essa função tem
-- 3 RemoteFunctions como upvalue, então aqui vai por índice + validação.
local searchRemote
if TradeController and getupvals and type(TradeController._createPlayerList) == "function" then
    local ok, ups = pcall(getupvals, TradeController._createPlayerList)
    local v = ok and type(ups) == "table" and ups[12]
    if typeof(v) == "Instance" and v:IsA("RemoteFunction") and v ~= inviteRemote then
        -- Nome legível: tem que terminar exatamente em "TradeService/SearchUser".
        -- Nome sem "/" (ofuscado): aceita pela posição do upvalue.
        local suffix = "TradeService/SearchUser"
        if v.Name:sub(-#suffix) == suffix or not v.Name:find("/", 1, true) then
            searchRemote = v
        end
    end
end

if not inviteRemote then
    setStatus("✕ Remote não encontrada.", 220, 60, 60)
end

-- ─── LÓGICA DO BOTÃO ──────────────────────────────────────────────────────────
local busy = false
local lastSearch = 0

local function resetBtn()
    sendBtn.Text   = "ENVIAR TRADE"
    sendBtn.Active = true
    busy = false
end

sendBtn.MouseButton1Click:Connect(function()
    if busy then return end
    local name = tb.Text:match("^%s*(.-)%s*$")

    if name == "" then
        setStatus("⚠ Digite um username!", 220, 80, 80, 3)
        return
    end

    if not inviteRemote then
        setStatus("✕ Remote não encontrada.", 220, 60, 60, 4)
        return
    end

    busy = true
    sendBtn.Text   = "Buscando..."
    sendBtn.Active = false
    setStatus("Buscando " .. name .. "...", 130, 130, 130)

    local ok, uid = pcall(function()
        return Players:GetUserIdFromNameAsync(name)
    end)

    if not ok or typeof(uid) ~= "number" then
        setStatus("✕ Jogador não encontrado.", 220, 60, 60, 3)
        resetBtn()
        return
    end

    if uid == LP.UserId then
        setStatus("✕ Não dá pra mandar trade pra você mesmo.", 220, 60, 60, 3)
        resetBtn()
        return
    end

    -- Mesmas condições do jogo: canInvite e inGame.
    -- Quem está no servidor: inGame = true, canInvite = true (TradeController:533).
    -- Quem está fora: pergunta ao servidor via SearchUser (TradeController:615).
    if not Players:GetPlayerByUserId(uid) then
        if not searchRemote then
            setStatus("✕ Jogador fora do servidor (SearchUser não encontrada).", 220, 60, 60, 4)
            resetBtn()
            return
        end
        local remaining = SEARCH_COOLDOWN - (os.clock() - lastSearch)
        if remaining > 0 then
            setStatus(("⏳ Aguarde %.0fs para buscar de novo."):format(remaining), 220, 160, 60, 3)
            resetBtn()
            return
        end
        lastSearch = os.clock()

        local sOk, found, inGame, canInvite = pcall(function()
            return searchRemote:InvokeServer(SEARCH_GUID, uid)
        end)
        if not (sOk and found) then
            local msg = typeof(found) == "string" and found
                or typeof(inGame) == "string" and inGame
                or "Falha ao buscar jogador."
            setStatus("✕ " .. msg, 220, 60, 60, 4)
            resetBtn()
            return
        end
        if not inGame then
            setStatus("✕ " .. name .. " não está no jogo.", 220, 60, 60, 4)
            resetBtn()
            return
        end
        if not canInvite then
            setStatus("✕ " .. name .. " não aceita trades.", 220, 60, 60, 4)
            resetBtn()
            return
        end
    end

    setStatus("Enviando invite...", 130, 130, 130)
    sendBtn.Text = "ENVIANDO..."

    -- Resultado final chega por TradeService/InviteResult, que o próprio
    -- TradeController escuta (Start, linha 2126) e repassa a este callback.
    local done = false
    local function onResult(delivered, message)
        if done then return end
        done = true
        if delivered then
            setStatus("✔ Invite enviado para " .. name, 60, 200, 100, 5)
        else
            setStatus("✕ " .. (typeof(message) == "string" and message or "Invite não entregue."), 220, 60, 60, 5)
        end
        resetBtn()
    end

    -- Chama a função do jogo: mesma remote, mesmo GUID, userId como number.
    local callOk, accepted, err = pcall(TradeController.SendInvite, TradeController, uid, onResult)

    if not callOk then
        onResult(false, "Falhou ao enviar invite.")
    elseif not accepted then
        onResult(false, err)
    else
        -- Servidor aceitou o pedido; esperando o InviteResult.
        task.delay(10, function()
            if not done then
                onResult(false, "Sem resposta do servidor.")
            end
        end)
    end
end)
