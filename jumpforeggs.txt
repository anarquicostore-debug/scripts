local getupvals = (debug and debug.getupvalues) or getupvalues
local getinfo   = debug and debug.getinfo

local function fnName(fn)
    if not getinfo then return nil end
    local ok, info = pcall(getinfo, fn)
    return ok and type(info) == "table" and info.name or nil
end

local function upvalues(fn)
    if not getupvals or type(fn) ~= "function" then return nil end
    local ok, ups = pcall(getupvals, fn)
    return ok and type(ups) == "table" and ups or nil
end

-- Única instância da classe entre os upvalues de fn.
local function onlyRemote(fn, class)
    local ups = upvalues(fn)
    if not ups then return nil end
    local found
    for _, v in pairs(ups) do
        if typeof(v) == "Instance" and v:IsA(class) then
            if found and found ~= v then return nil end
            found = v
        end
    end
    return found
end

-- Nome exato dentro das pastas do Net (só GetChildren).
local function remoteByName(path, class)
    local folders = {}
    local pk = ReplicatedStorage:FindFirstChild("Packages")
    if pk then
        if pk:FindFirstChild("Net") then table.insert(folders, pk.Net) end
        local idx = pk:FindFirstChild("_Index")
        if idx then
            for _, c in ipairs(idx:GetChildren()) do
                if c.Name:lower():sub(1, 13) == "sleitnick_net" and c:FindFirstChild("net") then
                    table.insert(folders, c.net)
                end
            end
        end
    end
    for _, folder in ipairs(folders) do
        for _, c in ipairs(folder:GetChildren()) do
            if c:IsA(class) and (c.Name == path or c.Name:sub(-(#path + 1)) == "/" .. path) then
                return c
            end
        end
    end
    return nil
end

local function onOwnPlot()
    local _, hrp = getChar()
    local plot = getOwnPlot()
    local floor = plot and plot:FindFirstChild("Floor")
    if not (hrp and floor and floor:IsA("BasePart")) then return false end

    return math.abs(p.X) <= floor.Size.X / 2 and math.abs(p.Z) <= floor.Size.Z / 2
end

-- isOverPad (linha 534)
local function isOverPad(pad, rel, margin, height)
    local half = pad.Size / 2
    if rel.Y < -half.Y or rel.Y > half.Y + height then return false end
    if pad:GetAttribute("JumpPadCircular") ~= true then
        return math.abs(rel.X) <= half.X + margin and math.abs(rel.Z) <= half.Z + margin
    end
    local r = math.min(half.X, half.Z) + margin
    return rel.X * rel.X + rel.Z * rel.Z <= r * r
end

local function nearestPad(hrp)
    local best, bestD
    for _, pad in ipairs(CollectionService:GetTagged("JumpPad")) do
        if pad:IsA("BasePart") and pad:IsDescendantOf(workspace) then
            local d = (pad.Position - hrp.Position).Magnitude
            if not bestD or d < bestD then best, bestD = pad, d end
        end
    end
    return best, bestD
end

-- ─── FLY ──────────────────────────────────────────────────────────────────────
-- Movimento físico com LinearVelocity (igual launchFromPad, linha 555) seguindo
-- uma rota calculada com Spherecast:
--   início → desvia pro lado até achar céu livre → sobe/desce reto até a
--   altitude de cruzeiro → vai de lado → desce ao lado/em cima do alvo → alvo
-- Proteções:
--   • rota testada com folga do tamanho do personagem; se não achar, tenta com
--     folga menor e, no fim, uma rota alta sem teste;
--   • olha à frente a cada frame e freia antes de bater em algo;
--   • se o servidor puxar de volta ou não houver progresso, baixa a velocidade e
--     recalcula (até 6 vezes) em vez de parar;
--   • qualquer erro limpa tudo (sem forças presas no personagem);
--   • se parar no ar (alvo sumiu, travou, erro, cancelado), leva o player pra
--     ilha mais próxima, ou pro próprio plot, em vez de deixar cair.
local FLY = { speed = 60, radius = 4 }
local flying = nil
local landing = false
local safeLand -- definida abaixo

local function castParams(extraIgnore)
    local p = RaycastParams.new()
    p.FilterType = Enum.RaycastFilterType.Exclude
    local ignore = {}
    if LP.Character then table.insert(ignore, LP.Character) end
    for _, x in ipairs(extraIgnore or {}) do
        if typeof(x) == "Instance" then table.insert(ignore, x) end
    end
    p.FilterDescendantsInstances = ignore
    p.RespectCanCollide = true
    p.IgnoreWater = true
    return p
end

local casts = 0
-- Spherecast dá erro acima de 1024 studs: testa em trechos de 900.
local function clear(a, b, params, radius)
    local d = b - a
    local len = d.Magnitude
    if len < 0.5 then return true end
    local dir, from, done = d.Unit, a, 0
    while done < len do
        local step = math.min(900, len - done)
        casts = casts + 1
        if casts % 250 == 0 then task.wait() end
        if workspace:Spherecast(from, radius, dir * step, params) then return false end
        from = from + dir * step
        done = done + step
    end
    return true
end

local function ringOffsets(radii, dirs)
    local list = { Vector3.zero }
    for _, r in ipairs(radii) do
        for i = 0, dirs - 1 do
            local ang = i / dirs * math.pi * 2
            table.insert(list, Vector3.new(math.cos(ang) * r, 0, math.sin(ang) * r))
        end
    end
    return list
end
local START_OFFSETS = ringOffsets({ 12, 25, 45, 70, 100, 140 }, 16)
local GOAL_OFFSETS  = ringOffsets({ 10, 20, 35, 55 }, 8)

local function planPath(start, goal, params, radius)
    casts = 0
    if math.abs(start.Y - goal.Y) < 8 and clear(start, goal, params, radius) then
        return { goal }
    end
    local alts = { goal.Y }
    local base = math.max(goal.Y, start.Y)
    for k = 1, 10 do table.insert(alts, base + k * 25) end
    for _, alt in ipairs(alts) do
        local outs = {}
        for _, off in ipairs(START_OFFSETS) do
            local A = start + off
            local B = Vector3.new(A.X, alt, A.Z)
            if clear(start, A, params, radius) and clear(A, B, params, radius) then
                table.insert(outs, { A = A, B = B })
                if #outs >= 6 then break end
            end
        end
        local ins = {}
        if #outs > 0 then
            for _, off in ipairs(GOAL_OFFSETS) do
                local D = goal + off
                local C = Vector3.new(D.X, alt, D.Z)
                if clear(C, D, params, radius) and clear(D, goal, params, radius) then
                    table.insert(ins, { C = C, D = D })
                    if #ins >= 6 then break end
                end
            end
        end
        for _, o in ipairs(outs) do
            for _, i in ipairs(ins) do
                if clear(o.B, i.C, params, radius) then
                    local wp = {}
                    for _, p in ipairs({ o.A, o.B, i.C, i.D, goal }) do
                        local last = wp[#wp] or start
                        if (p - last).Magnitude > 1 then table.insert(wp, p) end
                    end
                    log(("Rota: %d pontos, altitude %.0f, folga %.1f, %d testes"):format(#wp, alt, radius, casts))
                    return wp
                end
            end
        end
    end
    return nil
end

-- Tenta folgas menores e, por último, uma rota alta sem teste.
local function planWithFallback(start, goal, ignore)
    local params = castParams(ignore)
    for _, r in ipairs({ FLY.radius, 3, 2 }) do
        local ok, wp = pcall(planPath, start, goal, params, r)
        if ok and wp then return wp end
        if not ok then log("erro na rota (folga " .. r .. "):", wp) end
    end
    log("sem rota testada: usando rota alta de reserva")
    local alt = math.max(start.Y, goal.Y) + 200
    return { Vector3.new(start.X, alt, start.Z), Vector3.new(goal.X, alt, goal.Z), goal }
end

-- Chão sólido logo abaixo de pos (nil se não tiver).
local function groundBelow(pos, params, depth)
    local ok, r = pcall(function()
        return workspace:Raycast(pos, Vector3.new(0, -(depth or 1000), 0), params)
    end)
    if ok and r and r.Instance and r.Normal.Y > 0.6 then return r.Position end
end

local function plotLanding()

        return floor.Position + Vector3.new(0, floor.Size.Y / 2 + 4, 0)

end

-- Ilha mais próxima: chão direto embaixo, senão procura em anéis em volta.
local function findSafeGround(origin)
    local params = castParams()
    local g = groundBelow(origin, params, 400)
    if g then return g + Vector3.new(0, 4, 0) end
    for _, rad in ipairs({ 30, 60, 110, 180, 280, 420, 650, 1000 }) do
        local best, bestD
        for i = 0, 15 do
            local ang = i / 16 * math.pi * 2
            local p = origin + Vector3.new(math.cos(ang) * rad, 150, math.sin(ang) * rad)
            local hit = groundBelow(p, params, 2500)
            if hit then
                local d = (hit - origin).Magnitude
                if not bestD or d < bestD then best, bestD = hit, d end
            end
        end
        if best then return best + Vector3.new(0, 4, 0) end
    end
    return plotLanding()
end

local function inAirWithoutGround(hrp, hum)
    if hum.FloorMaterial ~= Enum.Material.Air then return false end
    return groundBelow(hrp.Position, castParams(), 40) == nil
end

local function cleanup(f)
    f.alive = false
    if f.conn then pcall(function() f.conn:Disconnect() end) end
    pcall(function() f.lv:Destroy() end)
    pcall(function() f.att:Destroy() end)
    local _, hrp = getChar()
    if hrp then pcall(function() hrp.AssemblyLinearVelocity = Vector3.zero end) end
end

-- reason: "ok", "lost" (alvo sumiu), "stuck", "timeout", "error", "nochar",
--         "cancel" (usuário), "replace" (outro voo começou)
local function stopFly(reason)
    local f = flying
    if not f then return end
    flying = nil
    cleanup(f)
    reason = reason or "cancel"
    if f.onDone then task.spawn(pcall, f.onDone, reason == "ok", reason) end
    -- parou no meio do nada: leva pro chão
    if not f.isLanding and reason ~= "ok" and reason ~= "replace" and reason ~= "nochar" then
        task.spawn(function() safeLand() end)
    end
end

-- getTarget: devolve a posição alvo (Vector3) ou nil se o alvo sumiu.
-- ignore: instâncias que não contam como obstáculo (ex.: o próprio ovo).
local function flyTo(getTarget, onDone, ignore, isLanding)
    stopFly("replace")
    local _, hrp = getChar()
    if not hrp then return false end
    local okT, firstGoal = pcall(getTarget)
    if not okT or typeof(firstGoal) ~= "Vector3" then return false end

    local att = Instance.new("Attachment")
    att.Name = "JFE_FlyAtt"
    att.Parent = hrp
    local lv = Instance.new("LinearVelocity")
    lv.Name = "JFE_Fly"
    lv.Attachment0 = att
    lv.VelocityConstraintMode = Enum.VelocityConstraintMode.Vector
    lv.RelativeTo = Enum.ActuatorRelativeTo.World
    lv.MaxForce = math.huge
    lv.VectorVelocity = Vector3.zero -- parado no ar enquanto calcula
    lv.Parent = hrp

    local f = {
        alive = true, lv = lv, att = att, onDone = onDone, isLanding = isLanding,
        wp = nil, idx = 1, tries = 0, mul = 1, moves = 0, ignore = ignore,
    }
    flying = f

    local function target()
        local ok, t = pcall(getTarget)
        if ok and typeof(t) == "Vector3" then return t end
    end

    local function replan()
        local _, h = getChar()
        local goal = target()
        if not h then return stopFly("nochar") end
        if not goal then return stopFly("lost") end
        f.wp = nil
        setStatus("Calculando rota...", 130, 130, 130)
        task.spawn(function()
            local ok, wp = pcall(planWithFallback, h.Position, goal, ignore)
            if not f.alive or flying ~= f then return end
            if not ok or not wp then
                log("erro na rota:", wp)
                return stopFly("error")
            end
            local total, prev = 0, h.Position
            for _, p in ipairs(wp) do total = total + (p - prev).Magnitude; prev = p end
            f.wp, f.idx, f.goal = wp, 1, goal
            f.best, f.bestAt, f.prevDist = nil, nil, nil
            f.deadline = os.clock() + total / math.max(FLY.speed * f.mul, 20) * 2 + 15
            setStatus(("Voando (%d pontos, vel. %d)"):format(#wp, math.floor(FLY.speed * f.mul)), 60, 200, 100)
        end)
    end

    -- não andou / foi puxado de volta: menos velocidade e nova rota
    local function struggle(why)
        f.tries = f.tries + 1
        f.mul = math.max(f.mul * 0.5, 40 / math.max(FLY.speed, 40))
        log(("%s — tentativa %d, velocidade agora %d"):format(why, f.tries, math.floor(FLY.speed * f.mul)))
        if f.tries > 6 then return stopFly("stuck") end
        replan()
    end

    local function step(dt)
        local _, h, hm = getChar()
        if not h or h ~= hrp or not lv.Parent then return stopFly("nochar") end
        local goal = target()
        if not goal then return stopFly("lost") end
        if not f.wp then
            lv.VectorVelocity = Vector3.zero
            return
        end
        if (goal - f.goal).Magnitude > 6 then
            f.goal = goal
            f.moves = f.moves + 1
            if f.moves > 15 then return stopFly("lost") end
            return replan()
        end
        if os.clock() > f.deadline then return stopFly("timeout") end

        local p = f.wp[f.idx]
        local delta = p - h.Position
        local dist = delta.Magnitude
        local last = f.idx == #f.wp
        if dist <= (last and 2 or 4) then
            if last then return stopFly("ok") end
            f.idx = f.idx + 1
            f.best, f.bestAt, f.prevDist = nil, nil, nil
            return
        end

        local dir = delta.Unit
        local spd = FLY.speed * f.mul
        -- não passa do ponto neste frame
        spd = math.min(spd, dist / math.max(dt, 1 / 240) * 0.8)
        -- olha à frente e freia antes de bater
        local look = math.min(dist, spd * dt * 4 + 6)
        local okC, hit = pcall(function()
            return workspace:Spherecast(h.Position, 2.5, dir * look, castParams(ignore))
        end)
        if okC and hit then
            if hit.Distance < 2 then
                lv.VectorVelocity = Vector3.zero
                return struggle("obstáculo à frente")
            end
            spd = math.min(spd, hit.Distance / math.max(dt, 1 / 240) * 0.4)
        end
        lv.VectorVelocity = dir * spd
        if hm then hm:ChangeState(Enum.HumanoidStateType.Freefall) end

        -- puxado de volta pelo servidor (distância aumentou de repente)
        if f.prevDist and dist > f.prevDist + 12 then
            f.prevDist = dist
            return struggle("servidor puxou de volta")
        end
        f.prevDist = dist

        -- sem progresso por 1,5s
        local now = os.clock()
        if not f.best or dist < f.best - 0.5 then
            f.best, f.bestAt = dist, now
        elseif now - f.bestAt > 1.5 then
            f.best, f.bestAt = nil, nil
            return struggle("sem progresso")
        end
    end

    f.conn = RunService.Heartbeat:Connect(function(dt)
        local ok, err = pcall(step, dt)
        if not ok then
            log("erro no voo:", err)
            stopFly("error")
        end
    end)

    replan()
    return true
end

-- Versão que espera: devolve (chegou, motivo).
local function flyWait(getTarget, ignore)
    local done, reached, reason = false, false, nil
    if not flyTo(getTarget, function(r, why) done, reached, reason = true, r, why end, ignore) then
        return false, "lost"
    end
    while not done do task.wait() end
    return reached, reason
end

-- Segura o personagem parado no ar (ex.: esperando o Grab). Devolve release().
local function holdPosition()
    local _, hrp = getChar()
    if not hrp then return function() end end
    local pos = hrp.Position
    local att = Instance.new("Attachment")
    att.Parent = hrp
    local lv = Instance.new("LinearVelocity")
    lv.Attachment0 = att
    lv.VelocityConstraintMode = Enum.VelocityConstraintMode.Vector
    lv.RelativeTo = Enum.ActuatorRelativeTo.World
    lv.MaxForce = math.huge
    lv.Parent = hrp
    local conn = RunService.Heartbeat:Connect(function()
        pcall(function()
            -- nunca briga com o voo: desliga enquanto outro voo estiver ativo
            lv.Enabled = flying == nil
            local d = pos - hrp.Position
            lv.VectorVelocity = d.Magnitude > 0.3 and d * 5 or Vector3.zero
        end)
    end)
    return function()
        pcall(function() conn:Disconnect() end)
        pcall(function() lv:Destroy() end)
        pcall(function() att:Destroy() end)
    end
end

-- Leva o player pra ilha mais próxima (ou pro plot) se estiver no ar sem chão.
safeLand = function()
    if landing or flying then return end
    local _, hrp, hum = getChar()
    if not hrp then return end
    local okAir, air = pcall(inAirWithoutGround, hrp, hum)
    if not okAir or not air then return end
    landing = true
    local release = holdPosition() -- não cai enquanto procura
    local okG, ground = pcall(findSafeGround, hrp.Position)
    release()
    if not okG or not ground then
        landing = false
        notify("Não achei chão perto. Voo liberado.", Color3.fromRGB(220, 60, 60))
        return
    end
    setStatus("Levando você pra ilha mais próxima...", 220, 160, 60)
    local started = flyTo(function() return ground end, function(ok, why)
        if ok or why == "replace" then
            landing = false
            if ok then setStatus("Pousou em segurança.", 60, 200, 100) end
            return
        end
        -- reserva: o próprio plot
        local plotPos = plotLanding()
        if plotPos and (plotPos - ground).Magnitude > 5 then
            flyTo(function() return plotPos end, function() landing = false end, nil, true)
        else
            landing = false
        end
    end, nil, true)
    if not started then landing = false end
end

local function distToPad(pad, pos)
    local rel = pad.CFrame:PointToObjectSpace(pos)
    local h = pad.Size / 2
    local c = Vector3.new(math.clamp(rel.X, -h.X, h.X), math.clamp(rel.Y, -h.Y, h.Y), math.clamp(rel.Z, -h.Z, h.Z))
    return (rel - c).Magnitude
end

local function readLevel()
    local lvl, cur, need, max
    pcall(function()
        local bar = LP.PlayerGui:FindFirstChild("JumpLTMLevel")
        local label = bar and bar:FindFirstChild("Label", true)
        if label and label:IsA("TextLabel") then
            local plain = label.Text:gsub("<[^>]->", "")
            lvl = tonumber(plain:match("Level%s*(%d+)"))
            local a, b = plain:match("(%d+)%s*/%s*(%d+)")
            cur, need, max = tonumber(a), tonumber(b), plain:find("MAX") ~= nil
        end
    end)
    local rawXp
    if syncData then
        pcall(function() rawXp = syncData:Get({ "JumpLTMEvent", "XP" }) end)
    end
    return lvl, cur, need, max, type(rawXp) == "number" and rawXp or nil
end

local function xpScore()
    local lvl, cur, _, _, raw = readLevel()
    if raw then return raw end
    if lvl then return lvl * 1e7 + (cur or 0) end
    return nil
end

local function esc(v)

local function hex(c)
    if typeof(c) ~= "Color3" then return "FFFFFF" end
    return string.format("%02X%02X%02X", math.floor(c.R * 255 + 0.5), math.floor(c.G * 255 + 0.5), math.floor(c.B * 255 + 0.5))

local function fmtTime(sec)
    if type(sec) ~= "number" then return nil end
    if sec <= 0 then return "READY!" end
    sec = math.ceil(sec)
    if sec >= 3600 then return string.format("%dh%02dm", sec // 3600, (sec % 3600) // 60) end
    return string.format("%d:%02d", sec // 60, sec % 60)
end
local function fmtVal(v)
    local t = typeof(v)
    if t == "number" then
        if math.abs(v) >= 1e6 then return string.format("%.3g", v) end
        return tostring(math.floor(v * 100 + 0.5) / 100)
    elseif t == "Color3" then return "#" .. hex(v)
    elseif t == "Vector3" then return string.format("%.0f, %.0f, %.0f", v.X, v.Y, v.Z)
    elseif t == "CFrame" then return string.format("%.0f, %.0f, %.0f", v.X, v.Y, v.Z)
    elseif t == "table" then
        local parts = {}
        for k, x in pairs(v) do
            local tx = typeof(x)
            if tx ~= "table" and tx ~= "function" and tx ~= "Instance" then
                table.insert(parts, (type(k) == "number" and "" or tostring(k) .. "=") .. tostring(x))
            end
            if #parts >= 6 then break end
        end
        return "{" .. table.concat(parts, ", ") .. "}"
    end
    return tostring(v)
end

-- ─── Coleta de ovos ──────────────────────────────────────────────────────────
local function plotOwner(inst)
    while inst and inst ~= workspace do
        if inst:IsA("Model") then
            local uid = inst:GetAttribute("OwnerUserId")
            if type(uid) == "number" then return uid end
        end
        inst = inst.Parent
    end
end

local function splitTraits(v)
    if type(v) == "table" then return v end
    if type(v) == "string" and v ~= "" then return string.split(v, "|") end
    return {}
end

-- Lê o overhead do jogo que fica dentro do model (fallback dos ovos do mapa).
local function readOverhead(model)
    local out = {}
    for _, g in ipairs(model:GetDescendants()) do
        if g:IsA("BillboardGui") or (g:IsA("GuiObject") and g:FindFirstChild("DisplayName")) then
            local root = g
            local dn = root:FindFirstChild("DisplayName", true)
            if dn and dn:IsA("TextLabel") then
                out.displayName = dn.Text
                local m = root:FindFirstChild("Mutation", true)
                if m and m:IsA("TextLabel") and m.Visible then out.mutation = m.Text end
                local r = root:FindFirstChild("Rarity", true)
                if r and r:IsA("TextLabel") then out.rarity = r.Text end
                local gen = root:FindFirstChild("Generation", true)
                if gen and gen:IsA("TextLabel") and gen.Visible then out.generation = gen.Text end
                local st = root:FindFirstChild("Stolen", true)
                if st and st:IsA("TextLabel") and st.Visible then out.stolen = st.Text end
                local tr = root:FindFirstChild("Traits", true)
                if tr then
                    out.traits = {}
                    for _, img in ipairs(tr:GetChildren()) do
                        if img:IsA("ImageLabel") and img.Visible and img.Name ~= "Template" then
                            table.insert(out.traits, traitByIcon[img.Image] or img.Image)
                        end
                    end
                end
                return out
            end
        end
    end
    return out
end

local function collectEggs()
    local list = {}
    -- Ovos colocados nos plots
    for _, m in ipairs(CollectionService:GetTagged("BasePlotEggVisual")) do
        -- ovos do meu próprio plot não aparecem
        if m:IsA("Model") and m:IsDescendantOf(workspace) and plotOwner(m) ~= LP.UserId then
            local attrs = m:GetAttributes()
            local uid = plotOwner(m)
            local owner = uid and Players:GetPlayerByUserId(uid)
            table.insert(list, {
                key = m, kind = "Plot", model = m,
                index = attrs.BrainrotIndex, mutation = attrs.Mutation,
                traits = splitTraits(attrs.Traits),
                hatchAt = attrs.HatchAt,
                owner = owner and owner.Name or (uid and ("UserId " .. uid)) or "?",
                attrs = attrs,
            })
        end
    end
    -- Ovos soltos no mapa (pool JumpLTM/Eggs)
    local rendered = JLC and type(JLC.RenderedEggs) == "table" and JLC.RenderedEggs or {}
    for id, rec in pairs(rendered) do
        local m = type(rec) == "table" and rec.model
        if typeof(m) == "Instance" and m:IsDescendantOf(workspace) then
            local data
            if eggPool then
                local ok, d = pcall(function() return eggPool:TryIndex({ "brainrots", id }) end)
                if ok and type(d) == "table" then data = d end
            end
            local oh = data and {} or readOverhead(m)
            local prompt = rec.prompt
            table.insert(list, {
                key = m, kind = "Mapa", model = m, id = id,
                index = data and data.brainrot or (typeof(prompt) == "Instance" and prompt.ObjectText) or oh.displayName,
                mutation = data and data.mutation or oh.mutation,
                traits = data and splitTraits(data.traits) or oh.traits or {},
                timer = data and data.timer, despawn = data and data.despawnTimer,
                grabbed = data and data.grabbed ~= nil or oh.stolen ~= nil,
                overhead = oh, data = data,
            })
        end
    end
    return list
end

-- ─── Texto de info ───────────────────────────────────────────────────────────
local function eggInfo(e, hrp)
    local a = type(e.index) == "string" and AnimalsData[e.index] or nil
    local name = a and a.DisplayName or e.index or "Egg"
    local rarity = a and a.Rarity or (e.overhead and e.overhead.rarity)
    local rd = rarity and RaritiesData[rarity]
    local rcolor = rd and rd.Color or Color3.fromRGB(230, 230, 230)

    local traitsTxt = {}
    for _, t in ipairs(e.traits) do table.insert(traitsTxt, traitRich(t)) end

    local timerTxt
    if type(e.hatchAt) == "number" then
        timerTxt = fmtTime(e.hatchAt - workspace:GetServerTimeNow())
    elseif type(e.timer) == "number" then
        timerTxt = fmtTime(e.timer)
    elseif e.overhead and e.overhead.generation then
        timerTxt = e.overhead.generation
    end

    local pos = e.model:GetPivot().Position
    local dist = hrp and (pos - hrp.Position).Magnitude

    local L = {}
    table.insert(L, ('<font size="15"><b>%s</b></font>'):format(esc(name)))
    if rarity then
        table.insert(L, ('Raridade: <font color="#%s"><b>%s</b></font>'):format(hex(rcolor), esc(rarity)))
    end
    table.insert(L, "Mutação: " .. (mutationRich(e.mutation) or "nenhuma"))
    table.insert(L, ("Traits (%d): %s"):format(#e.traits, #traitsTxt > 0 and table.concat(traitsTxt, ", ") or "nenhum"))
    -- o que o ovo choca (Datas.Animals[ovo].Egg.Animals)
    if a and type(a.Egg) == "table" and type(a.Egg.Animals) == "table" then
        local out = {}
        for _, x in ipairs(a.Egg.Animals) do
            if type(x) == "table" and x.Name then
                local ad = AnimalsData[x.Name]
                local r = ad and ad.Rarity and RaritiesData[ad.Rarity]
                table.insert(out, ('<font color="#%s">%s</font> (%s%%)'):format(
                    hex(r and r.Color), esc(ad and ad.DisplayName or x.Name), tostring(x.Chance or "?")))
            end
        end
        if #out > 0 then table.insert(L, "Choca: " .. table.concat(out, ", ")) end
        if a.Egg.Timer then table.insert(L, "Tempo de chocar: " .. esc(fmtTime(a.Egg.Timer))) end
    end
    if timerTxt then table.insert(L, "Choca em: <b>" .. esc(timerTxt) .. "</b>") end
    if type(e.despawn) == "number" then table.insert(L, "Some em: " .. esc(fmtTime(e.despawn))) end
    if e.kind == "Plot" then
        table.insert(L, "Plot de: <b>" .. esc(e.owner) .. "</b>")
    else
        table.insert(L, "No mapa" .. (e.grabbed and ' — <font color="#FF5555">PEGO</font>' or " — livre"))
    end
    if dist then table.insert(L, ("Distância: %.0f studs"):format(dist)) end
    table.insert(L, "Posição: " .. fmtVal(pos))
    if e.index and a and e.index ~= name then table.insert(L, "Index: " .. esc(e.index)) end

    -- Tudo o que tiver no Datas.Animals (campos simples)
    if a then
        local keys = {}
        for k, v in pairs(a) do
            local tv = typeof(v)
            if k ~= "DisplayName" and k ~= "Rarity" and tv ~= "function" and tv ~= "Instance" then
                table.insert(keys, k)
            end
        end
        table.sort(keys, function(x, y) return tostring(x) < tostring(y) end)
        for _, k in ipairs(keys) do
            table.insert(L, ('<font color="#9A9A9A">• %s: %s</font>'):format(esc(k), esc(fmtVal(a[k]))))
        end
    end
    -- Atributos do model / dados do pool
    local extra = e.attrs or e.data
    if extra then
        local keys = {}
        for k in pairs(extra) do table.insert(keys, k) end
        table.sort(keys, function(x, y) return tostring(x) < tostring(y) end)
        for _, k in ipairs(keys) do
            local v = extra[k]
            if k == "HatchAt" or k == "EggPlacedAt" or k == "PetGrowInAt" then
                v = type(v) == "number" and (fmtTime(v - workspace:GetServerTimeNow()) or v) or v
            end
            if typeof(v) ~= "function" then
                table.insert(L, ('<font color="#7A7A7A">◦ %s: %s</font>'):format(esc(k), esc(fmtVal(v))))
            end
        end
    end

    -- texto curto do ESP: nome + mutação colorida (os ícones vão em imagens)
    local short = "<b>" .. esc(name) .. "</b>" .. (e.mutation and (" " .. mutationRich(e.mutation)) or "")
    return table.concat(L, "\n"), short, rcolor, timerTxt, dist
end

-- land=false quando outro job vai começar logo em seguida.
function cancelJob(land)
    local j = currentJob
    currentJob = nil
    if j then
        j.cancelled = true
        if j.release then pcall(j.release) end
        resetButton(j)
    end
    local wasFlying = flying ~= nil
    stopFly(land == false and "replace" or "cancel")
    -- estava parado no ar (SOLTAR / pegando ovo): leva pro chão
    if land ~= false and not wasFlying then task.spawn(safeLand) end
end

local HIDDEN_POS = Vector3.new(10000, 10000, 10000)

local function eggTarget(model)
    return function()
        if not model.Parent or not model:IsDescendantOf(workspace) then return nil end
        local p = model:GetPivot().Position
        -- Ovo pego por alguém vai pra exatamente (10000, 10000, 10000)
        -- (WorldBrainrotController:111). Ovos da pista ficam bem alto (Y ~6500).
        if (p - HIDDEN_POS).Magnitude < 5 then return nil end
        return p + Vector3.new(0, 4, 0)
    end
end

local function waitAttr(name, value, timeout)
    local t0 = os.clock()
    while os.clock() - t0 < timeout do
        if LP:GetAttribute(name) == value then return true end
        task.wait(0.1)
    end
    return LP:GetAttribute(name) == value
end

local REASON_TEXT = {
    lost = "o alvo sumiu", stuck = "travou no caminho", timeout = "demorou demais",
    error = "erro no voo", nochar = "personagem não encontrado", cancel = "cancelado",
}

function startEggJob(e, card, button)
    cancelJob(false)
    local job = { card = card, button = button }
    currentJob = job
    button.Text = "PARAR"
    local function alive() return currentJob == job and not job.cancelled end
    local eggName = tostring(e.index or "Egg")

    local function finish(msg, ok, land)
        if job.release then pcall(job.release); job.release = nil end
        if currentJob == job then currentJob = nil end
        resetButton(job)
        if msg then setStatus(msg, ok and 60 or 220, ok and 200 or 60, ok and 100 or 60) end
        if land then task.spawn(safeLand) end
    end

    local function run()
        local model = e.model
        local tgt = eggTarget(model)
        if not tgt() then
            notify("" .. eggName .. " sumiu ou foi pego por alguém.", Color3.fromRGB(220, 60, 60))
            return finish("Ovo indisponível.", false, false)
        end

        setStatus("Voando até o ovo...", 130, 130, 130)
        local reached, why = flyWait(tgt, { model })
        if not alive() then return end
        if not reached then
            if why == "lost" then
                notify("" .. eggName .. " sumiu enquanto você ia até ele. Levando você pra ilha mais próxima.",
                    Color3.fromRGB(220, 60, 60))
            end
            -- stopFly já chama safeLand quando para no ar
            return finish("Não chegou no ovo: " .. (REASON_TEXT[why] or tostring(why)) .. ".", false, false)
        end

        -- Só voar: se não tiver chão, fica parado no ar até apertar SOLTAR
        if not (autoCollect and e.kind == "Mapa" and e.id ~= nil) then
            local _, hrp, hum = getChar()
            if hrp and inAirWithoutGround(hrp, hum) then
                job.release = holdPosition()
                button.Text = "SOLTAR"
                setStatus("Chegou no ovo (parado no ar). SOLTAR leva você pro chão.", 60, 200, 100)
                return -- o job continua ativo até SOLTAR/PARAR (cancelJob → safeLand)
            end
            return finish("Chegou no ovo.", true, false)
        end

        -- Pegar: mesmas condições do prompt do jogo
        -- (HoldDuration 1.25s e MaxActivationDistance 10, JumpLTMController:1248)
        if not grabRemote then return finish("Remote Grab não encontrada.", false, true) end
        if LP:GetAttribute("CarryingJumpEgg") ~= true then
            setStatus("Pegando o ovo...", 130, 130, 130)
            job.release = holdPosition() -- não cai enquanto segura o prompt
            task.wait(1.3)
            if not alive() then return end
            if not tgt() then
                notify("" .. eggName .. " sumiu antes de você pegar. Levando você pra ilha mais próxima.",
                    Color3.fromRGB(220, 60, 60))
                return finish("O ovo sumiu.", false, true)
            end
            local _, hrp = getChar()
            if not (hrp and (model:GetPivot().Position - hrp.Position).Magnitude <= 10) then
                return finish("Longe demais do ovo pra pegar.", false, true)
            end
            grabRemote:FireServer("JumpLTM/Eggs", e.id)
            log("Grab", e.id)
            local got = waitAttr("CarryingJumpEgg", true, 3)
            if job.release then job.release(); job.release = nil end
            if not alive() then return end
            if not got then return finish("Não conseguiu pegar o ovo.", false, true) end
        end

        -- Levar pro meu plot
        if not plotLanding() then return finish("Plot não encontrado.", false, true) end
        setStatus("Levando o ovo pro plot...", 130, 130, 130)
        reached, why = flyWait(plotLanding, {})
        if not alive() then return end
        if not reached then
            return finish("Não chegou no plot: " .. (REASON_TEXT[why] or tostring(why)) .. ".", false, false)
        end

        -- Colocar: mesma chamada do botão ButtonX do jogo (linha 1021)
        task.wait(0.3)
        if not onOwnPlot() then return finish("Não está em cima do plot.", false, true) end
        if not eggRemote then return finish("Remote RequestEggPlace não encontrada.", false, false) end
        eggRemote:FireServer()
        log("RequestEggPlace")
        if waitAttr("CarryingJumpEgg", false, 3) then
            notify("" .. eggName .. " colocado no seu plot!", Color3.fromRGB(60, 200, 100), 5)
            finish("Ovo colocado no plot!", true, false)
        else
            finish("Servidor não colocou o ovo (plot cheio?).", false, false)
        end
    end

    task.spawn(function()
        local ok, err = pcall(run)
        if not ok then
            log("erro no job:", err)
            notify("Erro no voo/coleta (veja F9). Levando você pro chão.", Color3.fromRGB(220, 60, 60))
            if flying then stopFly("error") end
            finish("Erro: " .. tostring(err), false, true)
        end
    end)
end

        if not img then return end
        local i = Instance.new("ImageLabel")
        i.Size = UDim2.new(0, 22, 0, 22)
        i.BackgroundTransparency = 1
        i.Image = tostring(img)
        i.Parent = icons
    end
    addIcon(mutationIcon(e.mutation))
    for _, t in ipairs(e.traits) do addIcon(traitIcon(t)) end

    return {
        card = card, cam = cam, center = center, radius = radius,
        info = info, stroke = cs, angle = math.random() * math.pi * 2,
    }
end

-- ─── LOOP ────────────────────────────────────────────────────────────────────
local entries = {} -- [model] = { e, esp, ui }

local function clearEntry(model)
    local ent = entries[model]
    if not ent then return end
    if ent.esp then ent.esp.hl:Destroy(); ent.esp.bb:Destroy() end
    if ent.ui then ent.ui.card:Destroy() end
    entries[model] = nil
end

local function refresh()
    local _, hrp = getChar()
    local list = collectEggs()
    local seen = {}
    for _, e in ipairs(list) do
        seen[e.key] = true
        local ent = entries[e.key]
        if not ent then
            ent = {}
            entries[e.key] = ent
        end
        ent.e = e
        local text, short, rcolor, timerTxt, dist = eggInfo(e, hrp)

        if toggles.esp then
            ent.esp = ent.esp or makeEsp(e)
            ent.esp.hl.FillColor = rcolor
            ent.esp.hl.OutlineColor = rcolor
            ent.esp.label.Text = ("%s\n%s%s"):format(short,
                timerTxt and (esc(timerTxt) .. "  ") or "",
                dist and ("%.0f studs"):format(dist) or "")
        elseif ent.esp then
            ent.esp.hl:Destroy(); ent.esp.bb:Destroy(); ent.esp = nil
        end

        if toggles.hub then
            ent.ui = ent.ui or makeCard(e)
            ent.ui.info.Text = text
            ent.ui.stroke.Color = rcolor
            ent.ui.card.LayoutOrder = math.floor(dist or 99999)
        elseif ent.ui then
            ent.ui.card:Destroy(); ent.ui = nil
        end
    end
    for model in pairs(entries) do
        if not seen[model] then clearEntry(model) end
    end
    local plot, mapa = 0, 0
    for _, e in ipairs(list) do if e.kind == "Plot" then plot = plot + 1 else mapa = mapa + 1 end end
    hubTitle.Text = ("Eggs no servidor: %d   (mapa %d · plots %d)"):format(#list, mapa, plot)
end

onEspToggle = function() refresh() end
onHubToggle = function(v) hub.Visible = v; refresh() end

task.spawn(function()
    while alive do
        if toggles.esp or toggles.hub then
            local ok, err = pcall(refresh)
            if not ok then log("refresh erro:", err) end
        end
        task.wait(0.5)
    end
    for model in pairs(entries) do clearEntry(model) end
end)

-- Troca a aranha 3D do botão lateral por um "V" (vale pra versão oficial da lib).
local function patchLauncher()
    local roots = {}
    pcall(function() table.insert(roots, gethui()) end)
    pcall(function() table.insert(roots, game:GetService("CoreGui")) end)
    for _, root in ipairs(roots) do
        local launcher = root:FindFirstChild("ChilliLeftCenter")
        local vp = launcher and launcher:FindFirstChild("ChilliViewport", true)
        if vp then
            for _, d in ipairs(vp:GetDescendants()) do
                if d:IsA("BasePart") then d.Transparency = 1 end
            end
            if not vp:FindFirstChild("V3ntIcon") then
                local v = Instance.new("TextLabel")
                v.Name = "V3ntIcon"
                v.AnchorPoint = Vector2.new(0.5, 0.5)
                v.Position = UDim2.fromScale(0.5, 0.5)
                v.Size = UDim2.fromScale(0.9, 0.9)
                v.BackgroundTransparency = 1
                v.Text = "V"
                v.TextScaled = true
                v.FontFace = Font.new("rbxasset://fonts/families/GothamSSm.json", Enum.FontWeight.Heavy)
                v.TextColor3 = Color3.new(1, 1, 1)
                v.ZIndex = vp.ZIndex + 1
                v.Parent = vp
                local st = Instance.new("UIStroke")
                st.Color, st.Thickness = Color3.new(0, 0, 0), 2
                st.Parent = v
            end
            local btn = vp.Parent
            local txt = btn and btn:FindFirstChild("Txt")
            if txt and txt:IsA("TextLabel") then txt.Text = "V3nt" end
            return true
        end
    end
    return false
end

local function loadChilli()
    -- 1º: cópia local com o "V" (pasta workspace do executor); 2º: versão oficial
    local sources = {}
    for _, name in ipairs(CHILLI_FILES) do
        local ok, content = pcall(function()
            if isfile and isfile(name) then return readfile(name) end
        end)
        if ok and type(content) == "string" and #content > 1000 then
            table.insert(sources, { name = name, src = content })
        end
    end
    table.insert(sources, { name = "oficial (GitHub)", get = function() return game:HttpGet(CHILLI_URL) end })

    for _, s in ipairs(sources) do
        local ok, lib = pcall(function()
            local src = s.src or s.get()
            if src:sub(1, 3) == "\239\187\191" then src = src:sub(4) end
            local chunk, err = loadstring(src)
            assert(chunk, err)
            local bootstrap = chunk()
            assert(type(bootstrap) == "function", "bootstrap inválido")
            local Chilli = bootstrap(CHILLI_KEY)
            assert(type(Chilli) == "table" and type(Chilli.CreateWindow) == "function", "API inválida")
            return Chilli
        end)
        log("Chilli (" .. s.name .. "):", ok and "ok" or tostring(lib))
        if ok then return lib end
    end
end

-- tenta os setters que a lib pode ter (o exemplo também não sabia qual existe)
local function setOptionText(opt, text)
    if type(opt) ~= "table" then return end
    for _, m in ipairs({ "SetText", "Set", "SetValue", "SetTitle", "Update" }) do
        if type(opt[m]) == "function" then
            if pcall(opt[m], opt, text) then return true end
        end
    end
    return false
end

        for _, ex in ipairs(tab.Sections or {}) do

        end
        return tab:CreateSection({ Name = name, Expanded = true })
    end

    local window = Chilli:CreateWindow({ Name = HUB_NAME, DefaultTab = "Main" })

                    if v then notify(name .. ": remote não encontrada no jogo.", Color3.fromRGB(220, 60, 60)) end
                    toggles[key] = false
                    return
                end
                toggles[key] = v == true
                if onChange then onChange(toggles[key]) end
            end),
        })
    end

    toggle(eggSec, "hub", "Hub de Eggs", "Lista de ovos com preview 3D e todas as informações.", true,
        function(v) if onHubToggle then onHubToggle(v) end end)

        end),

    flySec:CreateInput({
