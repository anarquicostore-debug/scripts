-- ─── RESOLVER DE REMOTES ──────────────────────────────────────────────────────
local getupvals = (debug and debug.getupvalues) or getupvalues
local getinfo   = debug and debug.getinfo

local function fnName(fn)
    if not getinfo then return nil end
    local ok, info = pcall(getinfo, fn)
    return ok and type(info) == "table" and info.name or nil
end

local function upvalues(fn)
    if not getupvals or type(fn) ~= "function" then return nil end
    local ok, ups = pcall(getupvals, fn)
    return ok and type(ups) == "table" and ups or nil
end

-- Única instância da classe entre os upvalues de fn.
local function onlyRemote(fn, class)
    local ups = upvalues(fn)
    if not ups then return nil end
    local found
    for _, v in pairs(ups) do
        if typeof(v) == "Instance" and v:IsA(class) then
            if found and found ~= v then return nil end
            found = v
        end
    end
    return found
end

-- Nome exato dentro das pastas do Net (só GetChildren).
local function remoteByName(path, class)
    local folders = {}
    local pk = ReplicatedStorage:FindFirstChild("Packages")
    if pk then
        if pk:FindFirstChild("Net") then table.insert(folders, pk.Net) end
        local idx = pk:FindFirstChild("_Index")
        if idx then
            for _, c in ipairs(idx:GetChildren()) do
                if c.Name:lower():sub(1, 13) == "sleitnick_net" and c:FindFirstChild("net") then
                    table.insert(folders, c.net)
                end
            end
        end
    end
    for _, folder in ipairs(folders) do
        for _, c in ipairs(folder:GetChildren()) do
            if c:IsA(class) and (c.Name == path or c.Name:sub(-(#path + 1)) == "/" .. path) then
                return c
            end
        end
    end
    return nil
end

local function resolve(label, fromFn, path, pick)
    local r, how
    if fromFn then
        r = pick and pick(fromFn) or onlyRemote(fromFn, "RemoteEvent")
        how = r and "upvalue"
    end
    if not r then r = remoteByName(path, "RemoteEvent"); how = r and "nome exato" end
    log(label, r and ("ok via " .. how .. " -> " .. r:GetFullName()) or "NÃO encontrada")
    return r
end

-- ─── HELPERS DO JOGO (cópias das checagens do controller) ─────────────────────
local function getChar()
    local c = LP.Character
    local hrp = c and c:FindFirstChild("HumanoidRootPart")
    local hum = c and c:FindFirstChildOfClass("Humanoid")
    if hrp and hum and hum.Health > 0 then return c, hrp, hum end
end

local function getOwnPlot()
    local plots = workspace:FindFirstChild("Plots")
    if not plots then return nil end
    for _, p in ipairs(plots:GetChildren()) do
        if p:IsA("Model") and p:GetAttribute("OwnerUserId") == LP.UserId then return p end
    end
end

-- isStandingOnOwnPlot (linha 1030)
local function onOwnPlot()
    local _, hrp = getChar()
    local plot = getOwnPlot()
    local floor = plot and plot:FindFirstChild("Floor")
    if not (hrp and floor and floor:IsA("BasePart")) then return false end
    local p = floor.CFrame:PointToObjectSpace(hrp.Position)
    return math.abs(p.X) <= floor.Size.X / 2 and math.abs(p.Z) <= floor.Size.Z / 2
end

-- isOverPad (linha 534)
local function isOverPad(pad, rel, margin, height)
    local half = pad.Size / 2
    if rel.Y < -half.Y or rel.Y > half.Y + height then return false end
    if pad:GetAttribute("JumpPadCircular") ~= true then
        return math.abs(rel.X) <= half.X + margin and math.abs(rel.Z) <= half.Z + margin
    end
    local r = math.min(half.X, half.Z) + margin
    return rel.X * rel.X + rel.Z * rel.Z <= r * r
end

local function nearestPad(hrp)
    local best, bestD
    for _, pad in ipairs(CollectionService:GetTagged("JumpPad")) do
        if pad:IsA("BasePart") and pad:IsDescendantOf(workspace) then
            local d = (pad.Position - hrp.Position).Magnitude
            if not bestD or d < bestD then best, bestD = pad, d end
        end
    end
    return best, bestD
end

local function makeToggle(key, label, y, enabled)
    local btn = Instance.new("TextButton")
    btn.Size, btn.Position = UDim2.new(1, -24, 0, 34), UDim2.new(0, 12, 0, y)
    btn.BorderSizePixel, btn.AutoButtonColor = 0, false

    btn.Parent = frame
    Instance.new("UICorner", btn).CornerRadius = UDim.new(0, 6)
    toggles[key] = false
    local function paint()
        if not enabled then
            btn.Text = label .. ": remote não encontrada"
            btn.BackgroundColor3 = Color3.fromRGB(40, 40, 40)
            btn.TextColor3 = Color3.fromRGB(120, 120, 120)
            return
        end
        btn.Text = label .. (toggles[key] and ": ON" or ": OFF")
        TweenService:Create(btn, TweenInfo.new(0.12), {
            BackgroundColor3 = toggles[key] and Color3.fromRGB(30, 140, 60) or Color3.fromRGB(170, 20, 20)
        }):Play()
    end
    btn.MouseButton1Click:Connect(function()
        if not enabled then return end
        toggles[key] = not toggles[key]
        paint()
    end)
    paint()
end

-- Farm: mesma lógica do PostSimulation do jogo (linhas 866-906):
-- cooldown 0.3s, velY <= 10, no chão ou caindo (velY <= -10), em cima do pad.
local launches, lastLaunch = 0, 0
local LAUNCH_CD = 0.35
table.insert(conns, RunService.Heartbeat:Connect(function()
    if not (toggles.farm and launchRemote) then return end
    local _, hrp, hum = getChar()
    if not hrp then return end
    local now = os.clock()
    if now - lastLaunch < LAUNCH_CD then return end

    local pad = nearestPad(hrp)
    if not pad then
        setStatus("Nenhum trampolim (tag JumpPad) encontrado.", 220, 60, 60)
        return
    end

    local height = math.max(4, hum.HipHeight + hrp.Size.Y)
    local grounded = hum.FloorMaterial ~= Enum.Material.Air
    local margin = grounded and 0 or 1
    local over = isOverPad(pad, pad.CFrame:PointToObjectSpace(hrp.Position), margin, height)

    if not over and toggles.tp then
        local top = pad.CFrame * CFrame.new(0, pad.Size.Y / 2 + hum.HipHeight + hrp.Size.Y / 2, 0)
        hrp.AssemblyLinearVelocity = Vector3.zero
        hrp.CFrame = CFrame.new(top.Position) * (hrp.CFrame - hrp.CFrame.Position)
        return -- espera o próximo frame pra checar de novo
    end

    local velY = hrp.AssemblyLinearVelocity.Y
    if over and velY <= 10 and (grounded or velY <= -10) then
        lastLaunch = now
        launchRemote:FireServer(pad)
        launches = launches + 1
        setStatus(("Farm: %d pulos em %s"):format(launches, pad:GetFullName():sub(-40)), 60, 200, 100)
    elseif not over then
        setStatus("Fique em cima do trampolim (ou ligue Teleportar).", 220, 160, 60)
    end
end))

-- Ovo / brainrot / upgrade: checa a cada 1s.
task.spawn(function()
    local lastUpgrade = 0
    while alive do
        task.wait(1)
        if not alive then break end

        -- Ovo: só se carregando (CarryingJumpEgg, linha 1004) e no próprio plot.
        if toggles.egg and eggRemote and LP:GetAttribute("CarryingJumpEgg") == true and onOwnPlot() then
            eggRemote:FireServer()
            log("RequestEggPlace")
            setStatus("Ovo colocado.", 60, 200, 100)
        end

        -- Brainrot: só se carregando (CarryingBaseBrainrot, linha 1069) e no
        -- próprio plot (o botão do jogo só fica ativo assim, linha 1083).
        if toggles.brainrot and brainRemote and LP:GetAttribute("CarryingBaseBrainrot") == true and onOwnPlot() then
            brainRemote:FireServer()
            log("RequestBrainrotPlace")
            setStatus("Brainrot colocado.", 60, 200, 100)
        end

        -- Upgrade: usa a placa do jogo (TrampolineSignGui, linha 1137): só envia
        -- se o botão Upgrade estiver visível e o LockOverlay (level) não.
        if toggles.upgrade and upgradeRemote and os.clock() - lastUpgrade >= 3 then
            local gui = LP:FindFirstChild("PlayerGui") and LP.PlayerGui:FindFirstChild("TrampolineSignGui")
            local up = gui and gui:FindFirstChild("Upgrade")
            local lock = gui and gui:FindFirstChild("LockOverlay")
            local plot = getOwnPlot()
            local order = plot and plot:GetAttribute("Order")
            if up and up.Visible and not (lock and lock.Visible) and order ~= nil then
                lastUpgrade = os.clock()
                upgradeRemote:FireServer(order)
                log("RequestTrampolineUpgrade", order)
                setStatus("Upgrade pedido (tier depende do dinheiro).", 60, 200, 100)
            end
        end
    end
end)
